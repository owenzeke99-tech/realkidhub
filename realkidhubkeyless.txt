-- Tên file lưu key ẩn trong thư mục workspace của Executor
local KEY_FILE = "RealKidHub_SavedKey.json"
local CORRECT_KEY = "RealKidHubKeyAdheoNdkwoebdkoejdkdndnfnfnjkshdnj"

-- Hàm kiểm tra xem key đã lưu còn hạn 24 giờ hay không
local function hasValidSavedKey()
    if not isfile or not readfile then return false end
    if not isfile(KEY_FILE) then return false end
    
    local success, data = pcall(function()
        return HttpService:JSONDecode(readfile(KEY_FILE))
    end)
    
    if success and data and data.key == CORRECT_KEY then
        local currentTime = os.time()
        -- 86400 giây = 24 giờ
        if currentTime - data.time < 86400 then
            return true
        end
    end
    return false
end

-- Hàm lưu key kèm theo mốc thời gian hiện tại
local function saveKey()
    if writefile then
        local data = {
            key = CORRECT_KEY,
            time = os.time()
        }
        pcall(function()
            writefile(KEY_FILE, HttpService:JSONEncode(data))
        end)
    end
end

-- Hàm chạy script chính
local function loadMainScript()
    local success, err = pcall(function()
        loadstring(game:HttpGet("https://raw.githubusercontent.com/realkidhub/realkid/refs/heads/main/main.lua"))()
    end)
    if not success then
        warn("Error loading main script: " + tostring(err))
    end
end

-- Nếu đã lưu key hợp lệ trong vòng 24 giờ trước đó, tự động vào game luôn không cần hiện bảng Key
